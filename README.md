# Task: Create a variable called 'for' and assign it a value 4

# 'for' is a reserved keyword in Python,
# so using it as a variable name will cause an error.

# Example:
# for = 4   # This will give SyntaxError

# Correct way:
for_value = 4

# Print the value
print("Value stored in variable:", for_value)

# Explanation
print("'for' cannot be used as a variable name because it is a reserved keyword in Python.")
